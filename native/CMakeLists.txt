cmake_minimum_required(VERSION 3.22)
project(kharchio_ai LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL ${IOS} CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_VULKAN ${ANDROID} CACHE BOOL "" FORCE)
set(GGML_OPENCL ${ANDROID} CACHE BOOL "" FORCE)
set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
include(FetchContent)
if(ANDROID)
  include(cmake/AndroidVulkan.cmake)
  include(cmake/AndroidOpenCL.cmake)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp/CMakeLists.txt")
  set(FETCHCONTENT_SOURCE_DIR_LLAMA "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/whisper.cpp/CMakeLists.txt")
  set(FETCHCONTENT_SOURCE_DIR_WHISPER "${CMAKE_CURRENT_SOURCE_DIR}/vendor/whisper.cpp")
endif()
# Exact revisions keep future builds reproducible. Existing vendor checkouts
# are reused; initial source provisioning is a development-time operation.
FetchContent_Declare(llama
  GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
  GIT_TAG d854d98bae598388ecea1ed84396587e5edc326d
  SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp")
FetchContent_Declare(whisper
  GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git
  GIT_TAG 231638a0717b7822784297e5980965ffcec05810
  SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/whisper.cpp")
FetchContent_MakeAvailable(llama whisper)
if(ANDROID AND TARGET vulkan-shaders-gen)
  include(ExternalProject)
  ExternalProject_Get_Property(vulkan-shaders-gen BINARY_DIR)
  set(HOST_BUILD_DIR "${BINARY_DIR}")
  set(NINJA_PATH "${CMAKE_MAKE_PROGRAM}")
  include(cmake/HostNinja.cmake)
endif()
if(IOS)
  add_library(kharchio_ai STATIC src/kharchio_ai.cpp src/whisper_qnn.cpp)
else()
  add_library(kharchio_ai SHARED src/kharchio_ai.cpp src/whisper_qnn.cpp)
endif()
target_link_libraries(kharchio_ai PRIVATE llama whisper)
if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a")
  set(_qairt "${CMAKE_CURRENT_SOURCE_DIR}/prebuilt/qairt/qairt/2.50.0.260828")
  if(EXISTS "${_qairt}/include/QNN/QnnInterface.h")
    target_include_directories(kharchio_ai PRIVATE "${_qairt}/include/QNN")
    target_compile_definitions(kharchio_ai PRIVATE KHARCHIO_HAS_QNN=1)
    target_link_libraries(kharchio_ai PRIVATE dl)
    message(STATUS "Kharchio Whisper: QAIRT 2.50 / Snapdragon 8 Gen 2 HTP v73")
  else()
    message(STATUS "Kharchio Whisper: run native/prepare_whisper_qnn.ps1 to package QNN; OpenCL/CPU fallback available")
  endif()
endif()
if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a")
  set(_hexagon_archive "${CMAKE_CURRENT_SOURCE_DIR}/prebuilt/hexagon/arm64-v8a/libggml-hexagon.a")
  if(EXISTS "${_hexagon_archive}")
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/prebuilt/hexagon/arm64-v8a/revision.txt" _hexagon_revision)
    if(NOT _hexagon_revision STREQUAL "d854d98bae598388ecea1ed84396587e5edc326d")
      message(FATAL_ERROR "Hexagon GGML revision mismatch. Run native/prepare_hexagon.ps1 again.")
    endif()
    add_library(kharchio_hexagon STATIC IMPORTED)
    set_target_properties(kharchio_hexagon PROPERTIES IMPORTED_LOCATION "${_hexagon_archive}")
    target_link_libraries(kharchio_ai PRIVATE kharchio_hexagon ggml-base dl)
    target_compile_definitions(kharchio_ai PRIVATE KHARCHIO_HAS_HEXAGON=1)
    message(STATUS "Kharchio: Qualcomm Hexagon backend included")
  else()
    message(STATUS "Kharchio: run native/prepare_hexagon.ps1 to include Hexagon; NPU mode will fall back")
  endif()
endif()
target_compile_options(kharchio_ai PRIVATE -fvisibility=hidden)
# Native inference must stay optimized even when Android Studio selects Debug.
# Flutter/Java debugging remains available; only the heavy AI kernels use -O3.
foreach(ai_target kharchio_ai whisper llama ggml ggml-base ggml-cpu ggml-vulkan ggml-opencl ggml-metal)
  if(TARGET ${ai_target} AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${ai_target} PRIVATE -O3)
    target_compile_definitions(${ai_target} PRIVATE NDEBUG)
  endif()
endforeach()
if(ANDROID)
  target_sources(kharchio_ai PRIVATE src/android_vulkan_compat.cpp)
  target_include_directories(kharchio_ai PRIVATE "${Vulkan_INCLUDE_DIR}")
  target_link_options(kharchio_ai PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
  target_link_libraries(kharchio_ai PRIVATE log android)
endif()
